#!/bin/bash
#
# Stage a fresh Chromebook (Crostini) container from nothing.
#
# This is the one piece that cannot live in the stage bucket, because the
# bucket is private and needs gcloud to read it -- and on a fresh container
# gcloud does not exist yet.  Keep a copy somewhere reachable without
# credentials (an unlisted Drive file, or a mail to yourself) and paste it
# in.  Everything after this script comes from the bucket.
#
# usage: bash stage [gs://bucket]          (or: BUCKET=gs://... bash stage)
#
# The bucket defaults to the stage bucket and may be overridden. It is not a
# secret: the bucket is private and every object in it is gated by IAM, so
# knowing its name gets you nothing without an authorised account. The login
# below is what protects the files.
#
# Idempotent: re-run it to repair a half-finished container.
#
# gcloud arrives in two steps.  First the tarball into $HOME: it needs no sudo
# and no keyring, so it works on a container that has nothing, and it is
# deliberately kept off PATH.  Then, once the bucket has been read and the
# machine is seeded, the apt-managed google-cloud-cli goes into /usr/bin --
# that is the copy the rest of the setup expects, and the one apt keeps up to
# date.  The tarball is removed once the packaged one answers.  Credentials
# live in ~/.config/gcloud and are shared by both, so you only ever log in
# once.

bucket=${1:-${BUCKET:-gs://stage.ticket.w5.glitnirticketing.com}}
sdk=$HOME/google-cloud-sdk
gcloud=$sdk/bin/gcloud

log()  { printf '\n\033[1;34m==>\033[0m %s\n' "$*"; }
warn() { printf '\033[1;33m[warn]\033[0m %s\n' "$*" >&2; }
die()  { printf '\033[1;31m[fail]\033[0m %s\n' "$*" >&2; exit 1; }

case $bucket in
  gs://*) ;;
  "")     die "empty bucket -- usage: bash stage [gs://bucket]" ;;
  *)      die "$bucket: expected a gs:// URL" ;;
esac

# --- gcloud, the tarball --------------------------------------------------
if command -v gcloud >/dev/null 2>&1 ; then
  gcloud=`command -v gcloud`
  log "gcloud: already on PATH -- `$gcloud version 2>/dev/null | head -1`"
elif [ -x "$gcloud" ] ; then
  log "gcloud: already installed at $sdk"
else
  # We only reach this when $sdk/bin/gcloud is missing, so anything still at
  # $sdk is a partial install -- an interrupted download or extract.  Clear
  # it: with --disable-prompts the installer refuses an existing directory
  # ("Remove ... or select a new installation directory, then run again")
  # rather than replacing it, which would wedge every retry from here on.
  if [ -e "$sdk" ] ; then
    log "gcloud: clearing a partial install at $sdk"
    rm -rf "$sdk" || die "cannot remove $sdk -- delete it by hand and re-run"
  fi
  log "gcloud: installing to $sdk (no sudo required)"
  tmp=`mktemp -t gcloud-install.XXXXXX` || die "cannot create temp file"
  trap 'rm -f "$tmp"' EXIT
  # -f so a 5xx or a captive portal is an error rather than an HTML page
  # that passes the -s test below and then gets handed to bash.
  curl -fsSL https://sdk.cloud.google.com > "$tmp" || die "cannot download installer"
  [ -s "$tmp" ] || die "installer downloaded empty"
  bash "$tmp" --disable-prompts --install-dir="$HOME" || die "gcloud install failed"
  rm -f "$tmp"
  trap - EXIT
  [ -x "$gcloud" ] || die "gcloud not found at $gcloud after install"
fi

# --- credentials ----------------------------------------------------------
# Opens a browser tab.  Crostini hands the URL to the ChromeOS browser, which
# is already signed in, so this is usually two clicks.
if "$gcloud" auth list --filter=status:ACTIVE --format='value(account)' 2>/dev/null | grep -q . ; then
  log "auth: active as `"$gcloud" auth list --filter=status:ACTIVE --format='value(account)' 2>/dev/null | head -1`"
else
  log "auth: no active account -- opening browser login"
  "$gcloud" auth login || die "gcloud auth login failed"
fi

# --- nnn ------------------------------------------------------------------
log "nnn: seeding ~/bin from the stage bucket"
mkdir -p "$HOME/bin" || die "cannot create $HOME/bin"
"$gcloud" storage cp "$bucket/nnn" "$HOME/bin/nnn" || die "cannot fetch nnn (no access to $bucket?)"
chmod 755 "$HOME/bin/nnn" || die "cannot chmod nnn"

# nnn needs gcloud on PATH; the tarball copy is not on it by design.
log "nnn: fetching scripts and dotfiles (a couple of minutes)"
PATH="`dirname "$gcloud"`:$PATH" "$HOME/bin/nnn" || die "nnn failed"

# --- gcloud, the apt one --------------------------------------------------
# Done last, once the container is seeded: this is the copy that lands on
# PATH and gets updated with everything else, so `bootstrap gcloud` finds it
# already in place.  Non-fatal -- the tarball still works and this is safe to
# re-run -- so a machine without sudo or without the network is staged all the
# same.
#
# Retry by re-running stage, not with `bootstrap gcloud`.  A failure here can
# leave the keyring behind, and `bootstrap gcloud` dearmors onto it without
# --yes, so it stops on "File exists".  The --yes below is why re-running
# stage does not.
apt_gcloud() {
  sudo apt-get update -qq || return 1
  # gnupg for the dearmor below; the rest is what apt needs to speak https to
  # a third-party repo.  Present on most images, not on a minimal one.
  sudo apt-get install -y -qq apt-transport-https ca-certificates curl gnupg || return 1
  sudo install -m 0755 -d /usr/share/keyrings || return 1
  # --yes so a re-run overwrites the keyring instead of stopping on it.
  curl -fsSL https://packages.cloud.google.com/apt/doc/apt-key.gpg \
    | sudo gpg --dearmor --yes -o /usr/share/keyrings/cloud.google.gpg || return 1
  echo "deb [signed-by=/usr/share/keyrings/cloud.google.gpg] https://packages.cloud.google.com/apt cloud-sdk main" \
    | sudo tee /etc/apt/sources.list.d/google-cloud-sdk.list >/dev/null || return 1
  sudo apt-get update -qq || return 1
  sudo apt-get install -y google-cloud-cli || return 1
}

case $gcloud in
  "$sdk"/*)
    if ! command -v sudo >/dev/null 2>&1 ; then
      warn "apt gcloud: no sudo on this container -- skipping, run \`bootstrap gcloud\` later"
    elif apt_gcloud ; then
      hash -r
      apt_bin=`command -v gcloud`
      case $apt_bin in
        ""|"$sdk"/*)
          warn "apt gcloud: installed but not on PATH -- keeping $sdk" ;;
        *)
          log "gcloud: now $apt_bin -- `"$apt_bin" version 2>/dev/null | head -1`"
          # The tarball was only ever a way in.  ~/.config/gcloud, which holds
          # the login, is a separate directory and is untouched.
          log "gcloud: removing the bootstrap tarball at $sdk"
          rm -rf "$sdk" || warn "could not remove $sdk -- delete it by hand" ;;
      esac
    else
      warn "apt gcloud: install failed -- keeping $sdk, retry with \`bash stage\`"
    fi
    ;;
  *)
    log "apt gcloud: gcloud already came from PATH ($gcloud) -- nothing to install" ;;
esac

cat <<'DONE'

  Staged.  Two commands left:

      exec bash -l      pick up the new ~/.bashrc, which defines `bootstrap`
      bootstrap         apt, gcloud, node, python, rust, go, docker, aws, agents

  gcloud is already the apt-managed one, so `bootstrap`'s gcloud stage is a
  no-op.  Then `bootstrap code` for the repos, and log out and back in once
  for the docker group.  Re-run `nnn` after `bootstrap code` to link
  make-resume and md2docx.py, which point into ~/resume.

DONE
